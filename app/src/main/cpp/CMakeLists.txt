cmake_minimum_required(VERSION 3.22.1)
project(nuvio_core LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set_property(GLOBAL PROPERTY JOB_POOLS native_compile=4)
set(CMAKE_JOB_POOL_COMPILE native_compile)
get_filename_component(REPOSITORY "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(CORE "${REPOSITORY}/shared/native-app")
set(DEPS "${REPOSITORY}/.native-deps")
set(SOURCES "${DEPS}/sources")
if(NOT EXISTS "${SOURCES}/quickjs/quickjs.c")
    message(FATAL_ERROR "Run python3 tools/native_dependencies.py before building the native core")
endif()
if(ANDROID)
    set(PREFIX "${DEPS}/prefix/${ANDROID_ABI}")
    foreach(lib curl ssl crypto)
        add_library(native_${lib} STATIC IMPORTED)
        set_target_properties(native_${lib} PROPERTIES IMPORTED_LOCATION "${PREFIX}/lib/lib${lib}.a")
    endforeach()
    set(NETWORK_LIBRARIES native_curl native_ssl native_crypto z dl)
    include_directories("${PREFIX}/include")
else()
    find_package(OpenSSL REQUIRED)
    find_package(CURL REQUIRED)
    set(NETWORK_LIBRARIES CURL::libcurl OpenSSL::SSL OpenSSL::Crypto z dl pthread)
endif()
set(EXPAT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(EXPAT_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SOURCES}/expat" "${CMAKE_BINARY_DIR}/expat")
set(UPNPC_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(UPNPC_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(UPNPC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(UPNPC_BUILD_SAMPLE OFF CACHE BOOL "" FORCE)
add_subdirectory("${SOURCES}/miniupnpc" "${CMAKE_BINARY_DIR}/miniupnpc")
add_library(quickjs STATIC "${SOURCES}/quickjs/quickjs.c" "${SOURCES}/quickjs/cutils.c"
    "${SOURCES}/quickjs/libregexp.c" "${SOURCES}/quickjs/libunicode.c" "${SOURCES}/quickjs/dtoa.c")
target_compile_definitions(quickjs PRIVATE _GNU_SOURCE CONFIG_VERSION="2026-06-04")
target_include_directories(quickjs PUBLIC "${SOURCES}/quickjs")
add_library(dht STATIC "${SOURCES}/dht/dht.c")
target_include_directories(dht PUBLIC "${SOURCES}")
# engine.cpp includes dht/dht.h; retain the upstream directory name.
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/include/dht")
configure_file("${SOURCES}/dht/dht.h" "${CMAKE_BINARY_DIR}/include/dht/dht.h" COPYONLY)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/include/miniupnpc")
file(GLOB UPNP_HEADERS "${SOURCES}/miniupnpc/include/*.h")
foreach(header ${UPNP_HEADERS})
    get_filename_component(name "${header}" NAME)
    configure_file("${header}" "${CMAKE_BINARY_DIR}/include/miniupnpc/${name}" COPYONLY)
endforeach()
set(CORE_SOURCES
    util stremio http nuvio_account watch_sync tracking connected_services connected_preparer
    providers trailers live_tv live_guide iptv_library media_server plex)
set(CORE_FILES "${CORE}/src/torrent/engine.cpp" "${CORE}/src/torrent/bencode.cpp"
    "${CORE}/gl/episode_playback.cpp" "${CMAKE_CURRENT_LIST_DIR}/core_bridge.cpp")
foreach(module ${CORE_SOURCES})
    list(APPEND CORE_FILES "${CORE}/src/${module}.cpp")
endforeach()
add_library(nuvio_shared STATIC ${CORE_FILES})
target_compile_definitions(nuvio_shared PRIVATE NUVIO_HAVE_PROVIDERS MINIUPNP_STATICLIB _FILE_OFFSET_BITS=64)
if(ANDROID)
    target_compile_definitions(nuvio_shared PRIVATE HAVE_DHT HAVE_UPNP)
else()
    target_compile_definitions(nuvio_shared PRIVATE NUVIO_TORRENT_LOCAL_FIXTURE)
endif()
target_include_directories(nuvio_shared PUBLIC "${CORE}/src" "${CORE}/gl" "${CORE}/ui"
    "${CORE}/third_party" "${CMAKE_BINARY_DIR}/include" "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(nuvio_shared PUBLIC quickjs dht miniupnpc::miniupnpc expat ${NETWORK_LIBRARIES})
if(ANDROID)
    add_library(nuvio_core SHARED jni_bridge.cpp)
    target_link_options(nuvio_core PRIVATE -Wl,-z,max-page-size=16384)
    target_link_libraries(nuvio_core PRIVATE nuvio_shared android log)
else()
    add_executable(nuvio_core_cli core_cli.cpp)
    target_link_libraries(nuvio_core_cli PRIVATE nuvio_shared)
endif()
